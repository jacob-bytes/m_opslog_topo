# ============================================================
# 主机与存储阵列配置
#
# 语法说明（大小写不敏感）：
#   [host 名称]        定义一台主机
#   [storage 名称]     定义一套存储阵列
#   port <端口名> = <WWN>   一个网口一行，WWN 全局唯一
#   port: <端口名> 再跟一行 wwn: <WWN> 也可以
#   ip / hostname / type 等键值均可选
# ============================================================

[host host-01]
ip = 10.20.1.11
port p1 = 10:00:00:00:c9:a1:00:01
port p2 = 10:00:00:00:c9:a1:00:02

[host host-02]
ip = 10.20.1.12
port p1 = 10:00:00:00:c9:a2:00:01
port p2 = 10:00:00:00:c9:a2:00:02

[host host-03]
hostname = host-03
ip = 10.20.1.13
wwn = 10:00:00:00:c9:a3:00:01

[host host-04]
ip = 10.20.1.14
port p1 = 10:00:00:00:c9:a4:00:01
port p2 = 10:00:00:00:c9:a4:00:02

# ------------------------------------------------------------
# 存储阵列：每套阵列有多个业务网口，分别连到不同交换机
# ------------------------------------------------------------

[storage storage-01]
ip = 10.20.1.31
port 0A = 20:00:00:00:c9:b1:00:01
port 0B = 20:00:00:00:c9:b1:00:02
port 1A = 20:00:00:00:c9:b1:00:03
port 1B = 20:00:00:00:c9:b1:00:04

[storage storage-02]
ip = 10.20.1.32
port 0A = 20:00:00:00:c9:b2:00:01
port 0B = 20:00:00:00:c9:b2:00:02
